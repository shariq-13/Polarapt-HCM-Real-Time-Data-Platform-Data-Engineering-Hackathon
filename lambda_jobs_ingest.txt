import os
import json
from datetime import datetime
import boto3
import urllib.request

s3 = boto3.client('s3')

def lambda_handler(event, context):
    bucket_name = os.environ['BUCKET_NAME']
    api_key = os.environ['USAJOBS_API_KEY']
    user_agent = os.environ['USER_AGENT']
    
    url = "https://data.usajobs.gov/api/search?ResultsPerBoundary=500"
    req = urllib.request.Request(url, headers={
        "Host": "data.usajobs.gov",
        "User-Agent": user_agent,
        "Authorization-Key": api_key
    })
    
    status_code = 200
    try:
        with urllib.request.urlopen(req) as response:
            data = json.loads(response.read().decode())
            record_count = len(data.get('SearchResult', {}).get('SearchResultItems', []))
            status_code = response.status
    except Exception as e:
        data = {"error": str(e)}
        record_count = 0
        status_code = 500

    now = datetime.utcnow()
    date_path = now.strftime("raw/jobs/%Y/%m/%d/%H%M.json")
    
    # Contract/metadata structure inside the file
    payload = {
        "contract": {
            "timestamp": now.isoformat(),
            "source": "jobs",
            "record_count": record_count,
            "response_status": status_code
        },
        "data": data
    }
    
    # Store object with S3 metadata contract
    s3.put_object(
        Bucket=bucket_name,
        Key=date_path,
        Body=json.dumps(payload),
        ContentType='application/json',
        Metadata={
            'timestamp': now.isoformat(),
            'source': 'jobs',
            'record_count': str(record_count),
            'response_status': str(status_code)
        }
    )
    return {"status": "Success", "records": record_count}
